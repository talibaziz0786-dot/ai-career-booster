export function useInterviewFeedback() {

  const navigate =
    useNavigate();

  const {

    category,

    difficulty,

    interviewType,

    language,

    history,

    answers,

    currentAnswer,

    resumeSkills,

    selectedSkills,

    feedbackLoading,

    setFeedbackLoading,

    setCareerInsights,

    setInterviewResult,

    updateHistoryEvaluation,

    endInterview,

  } = useInterviewStore();

  const generateInterviewResult =
    useCallback(async () => {

      if (
        feedbackLoading
      ) {
        return;
      }

      try {

        setFeedbackLoading(
          true
        );

        /* =====================================
           Answers
        ===================================== */

        const finalAnswers = [

          ...answers,

          ...(currentAnswer.trim()

            ? [currentAnswer]

            : []),

        ];

        /* =====================================
           History
        ===================================== */

        const finalHistory =
          history.map(
            (item) => ({
              ...item,
            })
          );

        if (

          currentAnswer.trim() &&

          finalHistory.length > 0

        ) {

          finalHistory[
            finalHistory.length - 1
          ] = {

            ...finalHistory[
              finalHistory.length - 1
            ],

            answer:
              currentAnswer,

          };

        }

        if (
          finalHistory.length === 0
        ) {

         toast.error(
  "Interview history is empty."
);

return;
        }

        if (

          finalHistory.some(
            (item) =>
              !item.answer?.trim()
          )

        ) {

          toast.error(
  "Please answer all questions before ending the interview."
);

return;

        }

        /* =====================================
           Feedback
        ===================================== */

        const feedback =
          await InterviewAPI.generateInterviewFeedback({

            history:
              finalHistory,

            answers:
              finalAnswers,

            language,

            skills:

              selectedSkills.length > 0

                ? selectedSkills

                : resumeSkills,

          });

        setInterviewResult(
          feedback.data
        );

        if (
          finalHistory.length > 0
        ) {

          updateHistoryEvaluation(

            finalHistory.length - 1,

            {

              score:
                feedback.data.overallScore,

              feedback:
                feedback.data.recruiterVerdict,

              strengths:
                feedback.data.strengths,

              weaknesses:
                feedback.data.weaknesses,

           Career Insights
        ===================================== */

        const insights =
          await InterviewAPI.generateCareerInsights({

            overallScore:
              feedback.data.overallScore,

            strengths:
              feedback.data.strengths,

            weaknesses:
              feedback.data.weaknesses,

          });

        setCareerInsights(
          insights.insights
        );

        /* =====================================
           Save Session
        ===================================== */

        await InterviewAPI.saveInterviewSession({

          category,

          difficulty,

          interviewType,

          language,

          skills:

            selectedSkills.length > 0

              ? selectedSkills

              : resumeSkills,

          history:
            finalHistory,

          overallScore:
            feedback.data.overallScore,

          recruiterVerdict:
            feedback.data.recruiterVerdict,

          strengths:
            feedback.data.strengths,

          weaknesses:
            feedback.data.weaknesses,

          recommendations:
            feedback.data.recommendations,

        });

        toast.success(
  "Interview completed successfully."
);

endInterview();

navigate(
  "/interview-result"
);

      } catch (error) {

       console.error(error);

toast.error(
  "Unable to generate interview report."
);

      } finally {

        setFeedbackLoading(

import { useInterviewStore } from "../../../store/interview-store";

/* ===========================================================
   Clone Interview History
=========================================================== */

export function cloneHistory() {
  return useInterviewStore
    .getState()
    .history.map((item) => ({
      ...item,
    }));
}

/* ===========================================================
   Attach Current Answer
=========================================================== */

export function attachCurrentAnswer(
  currentAnswer: string
) {
  const history = cloneHistory();

  if (
    currentAnswer.trim() &&
    history.length > 0
  ) {
    history[
      history.length - 1
    ] = {
      ...history[
        history.length - 1
      ],
      answer: currentAnswer,
    };
  }

  return history;
}

/* ===========================================================
   Validate History
=========================================================== */

export function validateHistory(
  history: ReturnType<
    typeof cloneHistory
  >
) {
  if (history.length === 0) {
    throw new Error(
      "Interview history is empty."
    );
  }

  const unanswered =
    history.some(
      (item) =>
        !item.answer?.trim()
    );

  if (unanswered) {
    throw new Error(
      "Please answer the current question before ending the interview."
    );
  }
}

/* ===========================================================
   Final Answers
=========================================================== */

export function buildFinalAnswers(
  answers: string[],
  currentAnswer: string
) {
  return [
    ...answers,

    ...(currentAnswer.trim()
      ? [currentAnswer]
      : []),
  ];
}

export const jobQueryKeys = {

/* ==========================================================

type JobCache =

/* ==========================================================
   Create Job Mutation
========================================================== */

export function useCreateJobMutation() {
  const queryClient =
    useQueryClient();

  return useMutation({
    mutationFn: (
      payload: CreateJobRequest
    ) =>
      JobAPI.createJob(
        payload
      ),

    /* ----------------------------------
       Optimistic Preparation
    ---------------------------------- */

    onMutate: async () => {

      await queryClient.cancelQueries({
        queryKey:
          jobQueryKeys.all,
      });

      const previousJobs =
        queryClient.getQueryData<JobCache>(
          jobQueryKeys.all
        );

      return {
        previousJobs,
      };
    },

    /* ----------------------------------
       Rollback
    ---------------------------------- */

    onError: (
      _error,
      _payload,
      context
    ) => {

      if (
        context?.previousJobs
      ) {
        queryClient.setQueryData<JobCache>(
          jobQueryKeys.all,
          context.previousJobs
        );
      }
    },

    /* ----------------------------------
       Server Success
    ---------------------------------- */

    onSuccess: async (
      createdJob
    ) => {

      /*
       * Keep individual detail cache
       * immediately available.
       */

      queryClient.setQueryData<JobItem>(
        jobQueryKeys.detail(
          createdJob.id
        ),
        createdJob
      );

      /*
       * Server is authoritative.
       * Refresh affected collections.
       */

      await invalidateJobListAndStats(
        queryClient
      );
    },

    /* ----------------------------------
       Final Synchronization
    ---------------------------------- */

    onSettled: async () => {

      await queryClient.invalidateQueries({
        queryKey:
          jobQueryKeys.all,
      });

      await queryClient.invalidateQueries({
        queryKey:
          jobQueryKeys.stats,
      });
    },
  });
}

   Update Job Mutation
========================================================== */

export function useUpdateJobMutation() {
  const queryClient =
    useQueryClient();

  return useMutation({
    mutationFn: ({
      id,
      payload,
    }: {
      id: string;

      payload: UpdateJobRequest;
    }) =>
      JobAPI.updateJob(
        id,
        payload
      ),

    /* ----------------------------------
       Optimistic Update
    ---------------------------------- */

    onMutate: async ({
      id,
      payload,
    }) => {

      await queryClient.cancelQueries({
        queryKey:
          jobQueryKeys.all,
      });

      await queryClient.cancelQueries({
        queryKey:
          jobQueryKeys.detail(
            id
          ),
      });

      const previousJobs =
        queryClient.getQueryData<JobCache>(
          jobQueryKeys.all
        );

      const previousJob =
        queryClient.getQueryData<JobItem>(
          jobQueryKeys.detail(
            id
          )
        );

      /*
       * Optimistically update list.
       */

      queryClient.setQueryData<JobCache>(
        jobQueryKeys.all,
        (
          oldJobs
        ) => {

          if (!oldJobs) {
            return oldJobs;
          }

          return oldJobs.map(
            (
              job
            ) =>
              job.id === id
                ? {
                    ...job,
                    ...payload,
                  }
                : job
          );
        }
      );

      /*
       * Optimistically update
       * individual detail cache.
       */

      if (previousJob) {

        queryClient.setQueryData<JobItem>(
          jobQueryKeys.detail(
            id
          ),
          {
            ...previousJob,
            ...payload,
          }
        );
      }

      return {
        previousJobs,

        previousJob,
      };
    },

    /* ----------------------------------
       Rollback
    ---------------------------------- */

    onError: (
      _error,
      variables,
      context
    ) => {

      if (
        context?.previousJobs
      ) {

        queryClient.setQueryData<JobCache>(
          jobQueryKeys.all,
          context.previousJobs
        );
      }

      if (
        context?.previousJob
      ) {

        queryClient.setQueryData<JobItem>(
          jobQueryKeys.detail(
            variables.id
          ),
          context.previousJob
        );
      }
    },

    /* ----------------------------------
       Server Success
    ---------------------------------- */

    onSuccess: async (
      updatedJob
    ) => {

      /*
       * Server response becomes
       * the source of truth.
       */

      queryClient.setQueryData<JobItem>(
        jobQueryKeys.detail(
          updatedJob.id
        ),
        updatedJob
      );

      await invalidateJobListAndStats(
        queryClient
      );
    },

    /* ----------------------------------
       Final Synchronization
    ---------------------------------- */

    onSettled: async (
      _data,
      _error,
      variables
    ) => {

      await Promise.all([
        queryClient.invalidateQueries({
          queryKey:
            jobQueryKeys.all,
        }),

        queryClient.invalidateQueries({
          queryKey:
            jobQueryKeys.stats,
        }),

        queryClient.invalidateQueries({
          queryKey:
            jobQueryKeys.detail(
              variables.id
            ),
        }),
      ]);
    },
  });
}

      id

        queryKey:
          jobQueryKeys.all,
      });

      const previousJobs =
        queryClient.getQueryData<JobCache>(
          jobQueryKeys.all
        );

       * Remove immediately from

            (
              job
            ) =>
              job.id !== id
          );
        }
      );

      return {
        previousJobs,
      };
    },

    /* ----------------------------------
       Rollback
    ---------------------------------- */

    onError: (
      _error,
      _id,
      context
    ) => {

      if (
        context?.previousJobs
      ) {

        queryClient.setQueryData<JobCache>(
          jobQueryKeys.all,
          context.previousJobs
        );

    },

    /* ----------------------------------
       Server Success
    ---------------------------------- */

    onSuccess: async (
      _data,
      id
    ) => {

      /*
       * Deleted job should no longer

      queryClient.removeQueries({
        queryKey:
          jobQueryKeys.detail(
            id
          ),
      });

      await invalidateJobListAndStats(
        queryClient

    /* ----------------------------------
       Final Synchronization
    ---------------------------------- */

    error,

    setLoading,

    setError,

    setResult,

    clearResult,

    resetProfile,
  } = useLinkedInStore();

  /* ===========================================================
     Generate AI Optimization
  =========================================================== */

  const optimizeProfile =
    useCallback(
      async (
        payload: Omit<
          LinkedInOptimizeRequest,
          "profile"
        >
      ) => {
        if (loading) {
          return;
        }

        try {
          setLoading(true);

          setError(null);

          clearResult();

          const response =
            await LinkedInAPI.optimizeLinkedInProfile(
              {
                profile,

                ...payload,
              }
            );

          if (!response.success) {
            throw new Error(
              response.message ||
                "Optimization failed."
            );
          }

          setResult(
            response.data
          );

          toast.success(
            "LinkedIn profile optimized successfully."
          );

          return response.data;
        } catch (err: any) {
          const message =
            err?.response?.data
              ?.message ||
            err?.message ||
            "Unable to optimize profile.";

          setError(message);

          toast.error(message);

          throw err;
        } finally {
          setLoading(false);
        }
      },
      [
        profile,

        loading,

        setLoading,

        setError,

        setResult,

        clearResult,
      ]
    );


      /* ===========================================================
     History
  =========================================================== */

  const loadHistory =
    useCallback(async () => {
      try {
        return await LinkedInAPI.getLinkedInHistory();
      } catch (error: any) {
        toast.error(
          error?.response?.data?.message ??
            "Unable to load history."
        );

        throw error;
      }
    }, []);

  /* ===========================================================
     Delete History
  =========================================================== */

  const deleteHistory =
    useCallback(async (id: string) => {
      try {
        await LinkedInAPI.deleteLinkedInHistory(
          id
        );

        toast.success(
          "History deleted successfully."
        );
      } catch (error: any) {
        toast.error(
          error?.response?.data?.message ??
            "Unable to delete history."
        );

        throw error;
      }
    }, []);

  /* ===========================================================
     Export PDF
  =========================================================== */

  const downloadPDF =
    useCallback(async (id: string) => {
      try {
        const blob =

        );

        throw error;
      }
    }, []);

    loading,
    setLoading,
    result,
    setResult,
  } = useResumeAnalysisStore();

  const analyze = async (
    resumeText: string,
    language = "english"
  ) => {
    try {
      setLoading(true);

import {
  useMutation,
  useQuery,
  useQueryClient,
} from "@tanstack/react-query";

import { toast } from "react-hot-toast";

import { getCurrentUser } from "../services/auth.service";
import {
  updateProfile,
  type UpdateProfilePayload,
  type UpdateProfileResponse,
} from "../services/user.service";

import {
  useAuthStore,
  type AuthUser,
} from "../store/auth-store";

export function useUserProfile() {
  const queryClient = useQueryClient();

  const user = useAuthStore(
    (state) => state.user
  );

  const setUser = useAuthStore(
    (state) => state.setUser
  );

  const {
    data,
    isLoading,
    error,
  } = useQuery<{ user: AuthUser }>({
    queryKey: ["currentUser"],

    queryFn: getCurrentUser,

    enabled: Boolean(user),

    staleTime: 1000 * 60 * 5,

    retry: 1,
  });

  /*
   * IMPORTANT:
   *
   * Do NOT continuously copy React Query data
   * into Zustand with useEffect.
   *
   * That can overwrite freshly updated profile data
   * with stale cached data.
   */

  const updateProfileMutation =
    useMutation<
      UpdateProfileResponse,
      Error,
      UpdateProfilePayload
    >({
      mutationFn: updateProfile,

      onSuccess: (responseData) => {
        const updatedUser =
          responseData.user;

        /*
         * 1. Update Zustand immediately.
         *
         * ProfileCard and ProfileCompletion
         * will re-render immediately.
         */
        setUser(updatedUser);

        /*
         * 2. Replace React Query cache immediately.
         *
         * We do NOT wait for another request.
         */
        queryClient.setQueryData(
          ["currentUser"],
          {
            user: updatedUser,
          }
        );

        /*
         * 3. Optional background revalidation.
         *
         * This verifies server state without
         * replacing the UI with stale data first.
         */
        void queryClient.invalidateQueries({
          queryKey: ["currentUser"],
          refetchType: "active",
        });

        toast.success(
          "Profile updated successfully!"
        );
      },

      onError: (error) => {
        toast.error(
          error.message ||
            "Failed to update profile."
        );
      },
    });

  /*
   * Prefer the React Query user when it exists,
   * otherwise use Zustand.
   */
  const currentUser =
    data?.user ?? user;

  return {
    user: currentUser,

    isLoading,

    error,

    updateProfile:
      updateProfileMutation.mutateAsync,

    isUpdating:
      updateProfileMutation.isPending,
  };
}

import axios from "axios";

import type {
  AxiosError,
  AxiosRequestConfig,
  InternalAxiosRequestConfig,
} from "axios";

/* ==========================================================
   API Configuration
========================================================== */

const API_URL =
  import.meta.env.VITE_API_URL;

if (!API_URL) {
  console.warn(
    "VITE_API_URL is not configured."
  );
}

/* ==========================================================
   Axios Instance
========================================================== */

export const api = axios.create({
  baseURL: API_URL,

  /**
   * Required for HTTP-only refresh cookies.
   */
  withCredentials: true,

  timeout: 30_000,
});

/* ==========================================================
   Extended Request Config
========================================================== */

interface RetryRequestConfig
  extends AxiosRequestConfig {
  _retry?: boolean;
}

/* ==========================================================
   Stored Auth Shape
========================================================== */

interface StoredAuth {
  state?: {
    token?: unknown;
  };
}

/* ==========================================================
   Auth Storage Helpers
========================================================== */

function getStoredAuth(): StoredAuth | null {
  try {
    const stored =
      localStorage.getItem(
        "careerboost-auth"
      );

    if (!stored) {
      return null;
    }

    const parsed: unknown =
      JSON.parse(stored);

    if (
      !parsed ||
      typeof parsed !== "object"
    ) {
      return null;
    }

    return parsed as StoredAuth;
  } catch {
    return null;
  }
}

/* ==========================================================
   Access Token
========================================================== */

function getAccessToken(): string | null {
  const auth =
    getStoredAuth();

  const token =
    auth?.state?.token;

  if (
    typeof token !== "string" ||
    token.trim().length === 0
  ) {
    return null;
  }

  return token.trim();
}

/* ==========================================================
   Save Access Token
========================================================== */

function saveAccessToken(
  token: string
): void {
  if (
    typeof token !== "string" ||
    token.trim().length === 0
  ) {
    return;
  }

  try {
    const auth =
      getStoredAuth();

    if (!auth) {
      return;
    }

    if (
      !auth.state ||
      typeof auth.state !== "object"
    ) {
      auth.state = {};
    }

    auth.state.token =
      token.trim();

    localStorage.setItem(
      "careerboost-auth",
      JSON.stringify(auth)
    );
  } catch {
    /**
     * Storage failure should not
     * crash the application.
     */
  }
}

/* ==========================================================
   Clear Local Auth
========================================================== */

function clearAuth(): void {
  try {
    localStorage.removeItem(
      "careerboost-auth"
    );
  } catch {
    // Ignore storage errors.
  }
}

/* ==========================================================
   Request Interceptor
========================================================== */

api.interceptors.request.use(
  (
    config: InternalAxiosRequestConfig
  ) => {
    const token =
      getAccessToken();

    if (token) {
      config.headers.Authorization =
        `Bearer ${token}`;
    }

    return config;
  },

  (error) =>
    Promise.reject(error)
);

/* ==========================================================
   Refresh Queue
========================================================== */

interface QueueItem {
  resolve: (
    token: string
  ) => void;

  reject: (
    error: unknown
  ) => void;
}

let isRefreshing = false;

let waitingQueue: QueueItem[] = [];

/* ==========================================================
   Process Queue
========================================================== */

function processQueue(
  error: unknown,
  token: string | null
): void {
  waitingQueue.forEach(
    ({
      resolve,
      reject,
    }) => {
      if (error) {
        reject(error);
        return;
      }

      if (!token) {
        reject(
          new Error(
            "Authentication refresh failed."
          )
        );

        return;
      }

      resolve(token);
    }
  );

  waitingQueue = [];
}

/* ==========================================================
   Refresh Access Token
========================================================== */

async function refreshAccessToken(): Promise<string> {
  if (!API_URL) {
    throw new Error(
      "VITE_API_URL is not configured."
    );
  }

  const response =
    await axios.post<{
      success: boolean;

      message?: string;

      token?: string;

      accessToken?: string;
    }>(
      `${API_URL}/api/auth/refresh`,
      {},
      {
        withCredentials: true,

        timeout: 30_000,
      }
    );

  const token =
    response.data.token ??
    response.data.accessToken;

  if (
    typeof token !== "string" ||
    token.trim().length === 0
  ) {
    throw new Error(
      response.data.message ??
        "Refresh response did not contain an access token."
    );
  }

  return token.trim();
}

/* ==========================================================
   Response Interceptor
========================================================== */

api.interceptors.response.use(
  (response) =>
    response,

  async (
    error: AxiosError
  ) => {
    const originalRequest =
      error.config as
        | RetryRequestConfig
        | undefined;

    /* ======================================================
       No Request Config
    ====================================================== */

    if (!originalRequest) {
      return Promise.reject(error);
    }

    /* ======================================================
       Only Handle 401
    ====================================================== */

    if (
      error.response?.status !== 401
    ) {
      return Promise.reject(error);
    }

    /* ======================================================
       Prevent Infinite Retry
    ====================================================== */

    if (
      originalRequest._retry
    ) {
      return Promise.reject(error);
    }

    /* ======================================================
       Never Refresh Refresh Endpoint
    ====================================================== */

    const requestUrl =
      originalRequest.url ?? "";

    if (
      requestUrl.includes(
        "/api/auth/refresh"
      )
    ) {
      return Promise.reject(error);
    }

    /* ======================================================
       Mark Request As Retried
    ====================================================== */

    originalRequest._retry = true;

    /* ======================================================
       Another Refresh Is Already Running
    ====================================================== */

    if (isRefreshing) {
      return new Promise(
        (
          resolve,
          reject
        ) => {
          waitingQueue.push({
            resolve: (
              token
            ) => {
              originalRequest.headers = {
                ...originalRequest.headers,

                Authorization:
                  `Bearer ${token}`,
              };

              resolve(
                api(originalRequest)
              );
            },

            reject,
          });
        }
      );
    }

    /* ======================================================
       Start Refresh
    ====================================================== */

    isRefreshing = true;

    try {
      const newToken =
        await refreshAccessToken();

      /* ====================================================
         Save New Access Token
      ==================================================== */

      saveAccessToken(
        newToken
      );

      /* ====================================================
         Resolve Waiting Requests
      ==================================================== */

      processQueue(
        null,
        newToken
      );

      /* ====================================================
         Retry Original Request
      ==================================================== */

      originalRequest.headers = {
        ...originalRequest.headers,

        Authorization:
          `Bearer ${newToken}`,
      };

      return api(
        originalRequest
      );
    } catch (refreshError) {
      /* ====================================================
         Reject Waiting Requests
      ==================================================== */

      processQueue(
        refreshError,
        null
      );

      /* ====================================================
         Clear Authentication
      ==================================================== */

      clearAuth();

      /* ====================================================
         Redirect To Login
      ==================================================== */

      if (
        typeof window !==
          "undefined" &&
        window.location.pathname !==
          "/login"
      ) {
        window.location.replace(
          "/login"
        );
      }

      return Promise.reject(
        refreshError
      );
    } finally {
      isRefreshing = false;
    }
  }
);

} from "@tanstack/query-sync-storage-persister";

import {
  queryClient,
} from "./react-query";

/* ==========================================================
   Persistence Configuration
========================================================== */

const CACHE_BUSTER =
  "careerboost-ai-v1";

/*
 * Persist query cache for 24 hours.
 *
 * After 24 hours the persisted cache is considered stale
 * and will be discarded.
 */
const MAX_CACHE_AGE =
  1000 *
  60 *
  60 *
  24;

/* ==========================================================
   Storage Persister
========================================================== */

export const queryPersister =
  createSyncStoragePersister({
    storage:
      typeof window !== "undefined"
        ? window.localStorage
        : undefined,

    key:
      "careerboost-ai-query-cache",

    throttleTime:
      1000,

    serialize:
      JSON.stringify,

    deserialize:
      JSON.parse,
  });

/* ==========================================================
   Start Persistence
========================================================== */

export function setupQueryPersistence() {
  if (
    typeof window ===
    "undefined"
  ) {
    return;
  }

  persistQueryClient({
    queryClient,

    persister:
      queryPersister,

    maxAge:
      MAX_CACHE_AGE,

    buster:
      CACHE_BUSTER,
  });
}

  QueryClient,
} from "@tanstack/react-query";

/* ==========================================================
   Query Client
========================================================== */

export const queryClient =
  new QueryClient({
    defaultOptions: {
      queries: {
        /*
         * Data remains fresh for 60 seconds.
         * During this period React Query will prefer
         * cached data instead of immediately refetching.
         */
        staleTime:
          1000 * 60,

        /*
         * Keep inactive queries in memory for 30 minutes.
         */
        gcTime:
          1000 * 60 * 30,

        /*
         * Refetch when browser reconnects.
         */
        refetchOnReconnect:
          true,

        /*
         * Background refresh when user returns
         * to the browser window.
         */
        refetchOnWindowFocus:
          true,

        /*
         * Retry failed requests.
         */
        retry: 2,

        /*
         * Small exponential retry delay.
         */
        retryDelay: (
          attemptIndex
        ) =>
          Math.min(
            1000 *
              2 **
                attemptIndex,
            30_000
          ),

        /*
         * Offline-aware query behaviour.
         *
         * The first request can use available cache,
         * while failed retries wait for connection.
         */
        networkMode:
          "offlineFirst",
      },

      mutations: {
        /*
         * Mutations also understand offline state.
         */
        networkMode:
          "offlineFirst",

        /*
         * Retry transient mutation failures.
         */
        retry: 1,

        retryDelay: 1000,
      },
    },
  });

import type {
  ResumeData,
  ResumeVersion,
} from "../store/resume-store";

const STORAGE_KEY =
  "careerboost-resume-history";

const MAX_HISTORY_VERSIONS = 10;

/*
 * Keep a controlled localStorage budget.
 *
 * We intentionally stay below the browser's
 * typical localStorage limit.
 */
const MAX_STORAGE_BYTES =
  3.5 * 1024 * 1024;

export interface LocalResumeVersion
  extends ResumeVersion {
  content: ResumeData;
  createdAt: string;
  updatedAt: string;
}

function generateId(): string {
  if (
    typeof crypto !== "undefined" &&
    typeof crypto.randomUUID === "function"
  ) {
    return crypto.randomUUID();
  }

  return `${Date.now()}-${Math.random()
    .toString(36)
    .slice(2)}`;
}

function calculateBytes(
  value: unknown
): number {
  return new Blob([
    JSON.stringify(value),
  ]).size;
}

function normalizeHistory(
  value: unknown
): LocalResumeVersion[] {
  if (!Array.isArray(value)) {
    return [];
  }

  return value.filter(
    (item): item is LocalResumeVersion =>
      Boolean(
        item &&
        typeof item === "object" &&
        typeof item._id === "string" &&
        typeof item.title === "string" &&
        typeof item.version === "number" &&
        typeof item.updatedAt === "string" &&
        typeof item.createdAt === "string" &&
        item.content &&
        typeof item.content === "object"
      )
  );
}

function readHistory(): LocalResumeVersion[] {
  try {
    const raw =
      localStorage.getItem(
        STORAGE_KEY
      );

    if (!raw) {
      return [];
    }

    return normalizeHistory(
      JSON.parse(raw)
    );
  } catch (error) {
    console.error(
      "Failed to read resume history:",
      error
    );

    return [];
  }
}

function writeHistory(
  history: LocalResumeVersion[]
): void {
  localStorage.setItem(
    STORAGE_KEY,
    JSON.stringify(history)
  );
}

export function getResumeHistory(): LocalResumeVersion[] {
  return readHistory();
}

export function getResumeHistoryById(
  id: string
): LocalResumeVersion | null {
  const history =
    readHistory();

  return (
    history.find(
      (resume) =>
        resume._id === id
    ) ?? null
  );
}

export function saveResumeSnapshot(
  data: ResumeData
): LocalResumeVersion {
  const existing =
    readHistory();

  const nextVersion =
    existing.length > 0
      ? Math.max(
          ...existing.map(
            (resume) =>
              resume.version
          )
        ) + 1
      : 1;

  const now =
    new Date().toISOString();

  /*
   * IMPORTANT:
   *
   * This is a COMPLETE resume snapshot.
   *
   * Personal information
   * Education
   * Experience
   * Projects
   * Skills
   * Photo
   * etc.
   *
   * Everything is restored exactly
   * when this version is opened.
   */
  const snapshot: LocalResumeVersion = {
    _id: generateId(),

    title:
      data.fullName?.trim()
        ? `${data.fullName.trim()} Resume`
        : "AI Resume",

    version: nextVersion,

    createdAt: now,

    updatedAt: now,

    content: {
      ...data,
    },
  };

  let updatedHistory = [
    snapshot,
    ...existing,
  ];

  /*
   * Keep maximum history versions.
   *
   * Do NOT silently delete old versions.
   */
  if (
    updatedHistory.length >
    MAX_HISTORY_VERSIONS
  ) {
    updatedHistory =
      updatedHistory.slice(
        0,
        MAX_HISTORY_VERSIONS
      );
  }

  const requiredBytes =
    calculateBytes(
      updatedHistory
    );

  if (
    requiredBytes >
    MAX_STORAGE_BYTES
  ) {
    throw new ResumeHistoryStorageError(
      "RESUME_HISTORY_STORAGE_LIMIT"
    );
  }

  try {
    writeHistory(
      updatedHistory
    );
  } catch (error) {
    console.error(
      "Failed to save resume history:",
      error
    );

    throw new ResumeHistoryStorageError(
      "RESUME_HISTORY_STORAGE_LIMIT"
    );
  }

  return snapshot;
}

export function deleteResumeSnapshot(
  id: string
): boolean {
  const history =
    readHistory();

  const updated =
    history.filter(
      (resume) =>
        resume._id !== id
    );

  if (
    updated.length ===
    history.length
  ) {
    return false;
  }

  writeHistory(updated);

  return true;
}

export function deleteOldestResumeSnapshot():
  LocalResumeVersion | null {
  const history =
    readHistory();

  if (history.length === 0) {
    return null;
  }

  /*
   * Newest version is index 0.
   * Oldest version is the last item.
   */
  const oldest =
    history[
      history.length - 1
    ];

  const updated =
    history.slice(0, -1);

  writeHistory(updated);

  return oldest;
}

export function clearResumeHistory(): void {
  localStorage.removeItem(
    STORAGE_KEY
  );
}

export function getResumeHistoryStorageInfo() {
  const history =
    readHistory();

  const bytes =
    calculateBytes(history);

  return {
    versions:
      history.length,

    maxVersions:
      MAX_HISTORY_VERSIONS,

    bytes,

    maxBytes:
      MAX_STORAGE_BYTES,

    usagePercent:
      Math.min(
        100,
        Math.round(
          (bytes /
            MAX_STORAGE_BYTES) *
            100
        )
      ),

    storageLimitReached:
      bytes >=
      MAX_STORAGE_BYTES,
  };
}

export class ResumeHistoryStorageError
  extends Error {
  code:
    | "RESUME_HISTORY_STORAGE_LIMIT";

  constructor(
    code:
      | "RESUME_HISTORY_STORAGE_LIMIT"
  ) {
    super(
      "Resume history storage limit reached."
    );

    this.name =
      "ResumeHistoryStorageError";

    this.code = code;
  }
}
